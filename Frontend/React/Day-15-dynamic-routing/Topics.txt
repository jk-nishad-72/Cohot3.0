*) Dynamic Routing -> 

     <Routes>
              <Route path='/' element ={<Home />} > </Route>
              <Route path='/about' element ={<About />} > </Route>
              <Route path='/products' element ={<Products />} > </Route>
              <Route path='/details/:id' element ={<ProductDetail />} > </Route>
    </Routes>
